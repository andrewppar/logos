** Useful nix functions

#+name: bash-fn
#+begin_src nix
  make-body = {commands}: builtins.foldl'
    (acc: elem: acc + " " + elem + "\n")
    ""
    commands ;

  shell-fn = {name, body}:
    "function " + name + "() {\n"
    + (make-body {commands = body;})
    + "\n};\n" ;
#+end_src
